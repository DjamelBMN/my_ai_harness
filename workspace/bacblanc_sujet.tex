\exercice[4]{1}{Thème : Probabilités conditionnelles et variables aléatoires}
Pour chacune des affirmations suivantes, indiquer si elle est vraie ou fausse. Chaque réponse devra être soigneusement justifiée ; une réponse non justifiée ne rapporte aucun point. Une absence de réponse ou une réponse inexacte n'enlève pas de point. Les questions de cet exercice sont indépendantes.
\vspace{0.55cm}
\begin{enumerate}
    \item Soit $A$ et $B$ deux événements indépendants d'une même probabilité $P$ tels que $P(A) = 0,4$ et $P(A \cup B) = 0,7$.\\[0.2cm]
    \textbf{Affirmation 1 :} La probabilité de l'événement $B$ est égale à $0,5$.
    \vspace{0.75cm}
    \item On considère une variable aléatoire $X$ suivant la loi binomiale $\mathcal{B}(10; 0,3)$.\\[0.2cm]
    \textbf{Affirmation 2 :} La probabilité de l'événement $\{X \ge 1\}$ est supérieure ou égale à $0,97$.
    \vspace{0.75cm}
    \item Soit $Z$ une variable aléatoire suivant la loi normale centrée réduite $\mathcal{N}(0;1)$.\\[0.2cm]
    \textbf{Affirmation 3 :} $P(-1,96 \le Z \le 1,96) \approx 0,95$.
    \vspace{0.75cm}
    \item Soit $Y$ une variable aléatoire suivant une loi exponentielle de paramètre $\lambda > 0$ telle que $P(Y \ge 2) = \mathrm{e}^{-1}$.\\[0.2cm]
    \textbf{Affirmation 4 :} Le paramètre $\lambda$ est égal à $0,5$.
\end{enumerate}

\exercice[5]{2}{Thème : Étude d'une fonction et calcul intégral}
On considère la fonction $g$ définie sur l'intervalle $]0; +\infty[$ par :
\[
    g(x) = x \ln(x) - x.
\]
On note $\mathcal{C}_g$ sa courbe représentative dans un repère orthonormé.
\vspace{0.25cm}
\partie{A --- Étude de la fonction $g$}
\begin{enumerate}
    \item Déterminer la limite de la fonction $g$ en $0$ (par valeurs supérieures).
    \item Déterminer la limite de la fonction $g$ en $+\infty$.
    \item Calculer la dérivée $g'(x)$ pour tout $x \in ]0; +\infty[$ et en déduire le tableau de variations de $g$.
    \item Démontrer que l'équation $g(x) = 0$ admet une unique solution $\alpha$ sur l'intervalle $[1; 3]$ et donner un encadrement d'amplitude $10^{-2}$ de $\alpha$.
\end{enumerate}
\vspace{0.55cm}
\partie{B --- Calcul d'aire et primitive}
\begin{enumerate}
    \item Démontrer que la fonction $G$ définie sur $]0; +\infty[0$ par $G(x) = \frac{1}{2}x^2\ln(x) - \frac{3}{4}x^2$ est une primitive de $g$ sur $]0; +\infty[$.
    \item Calculer l'aire exacte, en unité d'aire, du domaine compris entre la courbe $\mathcal{C}_g$, l'axe des abscisses et les droites d'équations $x = 1$ et $x = \mathrm{e}$.
\end{enumerate}

\exercice[6]{3}{Thème : Suites numériques et fonction associée}
On considère la suite $(u_n)$ définie par $u_0 = 2$ et, pour tout entier naturel $n$,
\[
    u_{n+1} = \frac{3u_n + 2}{u_n + 4}.
\]
\partie{A --- Étude d'une fonction auxiliaire}
\begin{enumerate}
    \item Soit $f$ la fonction définie sur $[0; +\infty[$ par $f(x) = \frac{3x+2}{x+4}$. Calculer $f'(x)$ et étudier son signe.
    \item Démontrer que pour tout $x \in [0; 2]$, $f(x) \in [0; 2]$.
    \item 
    \begin{enumerate}
        \item Résoudre dans $[0; +\infty[$ l'équation $f(x) = x$.
        \item En déduire les points fixes de la fonction $f$.
    \end{enumerate}
    \item Démontrer que pour tout entier naturel $n$, on a : $0 \le u_n \le 2$.
\end{enumerate}
\partie{B --- Monotonie et convergence de la suite}
\begin{enumerate}
    \item Démontrer que pour tout $n \in \mathbb{N}$, $u_{n+1} - u_n = \frac{(2-u_n)(u_n+1)}{u_n+4}$.
    \item En déduire le sens de variation de la suite $(u_n)$.
    \item Démontrer que la suite $(u_n)$ est convergente.
    \item Déterminer la limite $\ell$ de la suite $(u_n)$.

\exercice[5]{4}{Thème : Géométrie dans l'espace}
L'espace est muni d'un repère orthonormé $(O; \vec{i}, \vec{j}, \vec{k})$.
On considère les points $A(1; 0; 2)$, $B(0; 2; 1)$, $C(2; 1; 0)$ et $D(3; 3; 3)$.
\vspace{0.45cm}
\noindent\textbf{\large Partie A --- Nature du tétraèdre et plans}
\vspace{0.25cm}
\begin{enumerate}
    \item Démontrer que les points $A$, $B$ et $C$ ne sont pas alignés, et définir ainsi un plan $\mathcal{P}$.
    \item Démontrer qu'une équation cartésienne du plan $\mathcal{P}$ est $x + y + z - 3 = 0$.
    \item Vérifier si le point $D$ appartient au plan $\mathcal{P}$.
    \item Calculer le volume du tétraèdre $ABCD$.
\end{enumerate}
\vspace{0.55cm}
\noindent\textbf{\large Partie B --- Droites et orthogonalité}
\vspace{0.25cm}
\begin{enumerate}
    \item Déterminer une représentation paramétrique de la droite $\Delta$ passant par $D$ et orthogonale au plan $\mathcal{P}$.
    \item Déterminer les coordonnées du projeté orthogonal $H$ du point $D$ sur le plan $\mathcal{P}$.
    \item Calculer la distance du point $D$ au plan $\mathcal{P}$.
    \item Soit $\mathcal{S}$ la sphère de centre $D$ et de rayon $R = 3$. Déterminer la nature de l'intersection de la sphère $\mathcal{S}$ et du plan $\mathcal{P}$.
    \item 
    \begin{enumerate}[topsep=0.15cm, itemsep=0.3em]
        \item Justifier l'existence d'une intersection non vide.
        \item Calculer le rayon du cercle intersection.
    \end{enumerate}
