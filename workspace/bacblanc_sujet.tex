\vspace{0.35cm}\noindent\textbf{\large Résumé de la structure de l'épreuve}\par\vspace{0.15cm}

\begin{enumerate}
  \item \textbf{Exercice 1 : Analyse et suites numériques (4 points)}
  \item Étude d'une suite récurrente $u_{n+1} = \frac{3u_n+1}{u_n+3}$ avec $u_0 = 0$.
  \item Calculs de termes exacts, démonstration par récurrence, étude des variations de la fonction associée et de la suite, convergence par le théorème de convergence monotone et détermination de la limite exacte $\ell = 1$.
\end{enumerate}

\begin{enumerate}
  \item \textbf{Exercice 2 : Géométrie dans l'espace (5 points)}
  \item Étude d'un plan défini par trois points $A, B, C$ dans un repère orthonormé $(O\,;\,\vec{i},\,\vec{j},\,\vec{k})$.
  \item Détermination d'un vecteur normal, équation cartésienne du plan $(ABC) : x + y + 2z - 5 = 0$, représentation paramétrique d'une droite, calcul de distance d'un point au plan et calcul du volume du tétraèdre $DABC$.
\end{enumerate}

\begin{enumerate}
  \item \textbf{Exercice 3 : Probabilités et variables aléatoires (6 points)}
  \item Modélisation par un arbre pondéré (contrôle qualité de composants électroniques).
  \item Probabilités conditionnelles, formule des probabilités totales ($P(T) = 0,875$).
  \item Loi de probabilité d'une variable aléatoire $X$ donnant le bénéfice algébrique, calcul de l'espérance mathématique ($E(X) = 11,875$ €).
  \item Échantillonnage et loi binomiale $\mathcal{B}(50\,; 0,9)$ avec calculs de probabilités cumulées.
\end{enumerate}

\begin{enumerate}
  \item \textbf{Exercice 4 : Fonctions, logarithme et convexité (5 points)}
  \item Étude d'une fonction auxiliaire $g(x) = x^2 - 1 + \ln(x)$, application du corollaire du TVI pour l'unicité de la solution $\alpha$.
  \item Étude d'une fonction principale $f(x) = x - \frac{\ln(x)}{x}$, limites, asymptotes (oblique $y = x$), dérivée liée à $g(x)$, variations, point d'inflexion et tangente.
\end{enumerate}
