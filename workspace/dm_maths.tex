\section*{Exercice 1 : Étude théorique d'une suite arithmético-géométrique (6 points)}
\begin{enumerate}
  \item \textbf{Objectif} : Consolider les automatismes fondamentaux sur la manipulation des suites non arithmétiques et non géométriques en introduisant une suite auxiliaire.
  \item \textbf{Partie A (4 points)} : Étude de la suite $(u_n)$ définie par $u_0 = 2$ et $u_{n+1} = \frac{1}{2}u_n + 3$. Calcul des premiers termes, mise en évidence de l'absence de raison arithmétique ou géométrique globale, puis introduction de la suite auxiliaire $v_n = u_n - 6$. Démonstration que $(v_n)$ est géométrique de raison $q = \frac{1}{2}$ et expression explicite de $u_n$ en fonction de $n$.
  \item \textbf{Partie B (2 points)} : Analyse du comportement asymptotique (limite en $+\infty$) et résolution d'une inéquation de seuil pour déterminer à partir de quel rang $n$ on a $u_n > 5,99$.
\end{enumerate}

\section*{Exercice 2 : Modélisation financière et algorithmique (10 points)}
\begin{enumerate}
  \item \textbf{Objectif} : Appliquer les suites arithmético-géométriques à un problème concret de placement bancaire avec versements périodiques, combiné avec de l'algorithmique en Python.
  \item \textbf{Partie A (8 points)} : Modélisation de l'évolution d'un plan d'épargne avec un versement initial de $1000$ € et des versements mensuels de $150$ € rémunérés au taux mensuel de $0,4\%$ ($C_{n+1} = 1,004 C_n + 150$). Introduction de la suite auxiliaire $V_n = C_n + 37500$, détermination de son expression explicite, puis résolution d'un objectif d'épargne d'au moins $25000$ €.
  \item \textbf{Partie B (2 points)} : Analyse et complétion d'une fonction Python de calcul de seuil itératif (\texttt{while}), puis mise en relation des résultats algorithmiques avec la théorie mathématique précédente.
\end{enumerate}

\section*{Correction de l'Exercice 1}

\begin{enumerate}
  \item \textbf{Calculs et nature (1 pt)} :
  \begin{itemize}
    \item $u_1 = \frac{1}{2}u_0 + 3 = \frac{1}{2}(2) + 3 = 4$.
    \item $u_2 = \frac{1}{2}u_1 + 3 = \frac{1}{2}(4) + 3 = 5$.
    \item Différences : $u_1 - u_0 = 4 - 2 = 2$ et $u_2 - u_1 = 5 - 4 = 1$. Comme $u_1 - u_0 \neq u_2 - u_1$, la suite $(u_n)$ n'est \textbf{pas arithmétique}.
    \item Quotients : $\frac{u_1}{u_0} = \frac{4}{2} = 2$ et $\frac{u_2}{u_1} = \frac{5}{4} = 1,25$. Comme $\frac{u_1}{u_0} \neq \frac{u_2}{u_1}$, la suite n'est \textbf{pas géométrique}.
  \end{itemize}

  \item \textbf{Suite auxiliaire $(v_n)$ (3 pts)} :
  \begin{enumerate}[label=\alph*.]
    \item Pour tout $n \in \mathbb{N}$ :
  \end{enumerate}
\end{enumerate}
$$v_{n+1} = u_{n+1} - 6 = \left(\frac{1}{2}u_n + 3\right) - 6 = \frac{1}{2}u_n - 3 = \frac{1}{2}(u_n - 6) = \frac{1}{2}v_n$$
La suite $(v_n)$ est donc une suite \textbf{géométrique de raison $q = \frac{1}{2}$}.
Son premier terme est $v_0 = u_0 - 6 = 2 - 6 = -4$. \textit{(1,5 pt)}

\begin{enumerate}
  \setcounter{enumi}{1}
  \begin{enumerate}[label=\alph*.]
    \setcounter{enumii}{1}
    \item On en déduit l'expression de $v_n$ : $v_n = v_0 \times q^n = -4 \left(\frac{1}{2}\right)^n$.
  \end{enumerate}
\end{enumerate}
Puisque $v_n = u_n - 6$, on a $u_n = v_n + 6$, soit :
$$u_n = 6 - 4\left(\frac{1}{2}\right)^n$$ \textit{(1,5 pt)}

\begin{enumerate}
  \setcounter{enumi}{2}
  \item \textbf{Convergence et seuil (2 pts)} :
  \begin{enumerate}[label=\alph*.]
    \item Comme $-1 < \frac{1}{2} < 1$, on a $\lim_{n \to +\infty} \left(\frac{1}{2}\right)^n = 0$. Par opérations sur les limites, $\lim_{n \to +\infty} u_n = 6 - 4(0) = 6$. \textit{(1 pt)}
    \item Résolvons $u_n > 5,99$ :
  \end{enumerate}
\end{enumerate}
$$6 - 4\left(\frac{1}{2}\right)^n > 5,99 \iff 0,01 > 4\left(\frac{1}{2}\right)^n \iff \left(\frac{1}{2}\right)^n < 0,0025$$
En passant par la fonction logarithme strictement croissante :
$$n \ln\left(\frac{1}{2}\right) < \ln(0,0025) \iff n > \frac{\ln(0,0025)}{\ln(0,5)} \approx \frac{-5,9914}{-0,6931} \approx 8,64$$
Le plus petit entier $n$ recherché est donc \textbf{$n = 9$}. \textit{(1 pt)}

\section*{Correction de l'Exercice 2}

\begin{enumerate}
  \item \textbf{Modélisation (2 pts)} :
\end{enumerate}
Le capital au mois $n+1$ correspond au capital du mois $n$ augmenté de ses intérêts au taux de $0,4\%$ (soit multiplié par $1 + 0,004 = 1,004$), auquel on ajoute le versement fixe de $150$ € :
$$C_{n+1} = 1,004 \, C_n + 150 \quad \text{avec } C_0 = 1000$$

\begin{enumerate}
  \setcounter{enumi}{1}
  \item \textbf{Suite auxiliaire $(V_n)$ (2,5 pts)} :
\end{enumerate}
$$V_{n+1} = C_{n+1} + 37500 = (1,004 \, C_n + 150) + 37500 = 1,004 \, C_n + 37650$$
Exprimons $37650$ en fonction de $37500$ : $1,004 \times 37500 = 37650$. Ainsi :
$$V_{n+1} = 1,004 \, C_n + 1,004 \times 37500 = 1,004(C_n + 37500) = 1,004 \, V_n$$
La suite $(V_n)$ est géométrique de \textbf{raison $q = 1,004$} et de premier terme $V_0 = C_0 + 37500 = 1000 + 37500 = 38500$.

\begin{enumerate}
  \setcounter{enumi}{2}
  \item \textbf{Expression explicite de $C_n$ (2 pts)} :
\end{enumerate}
On a $V_n = V_0 \times q^n = 38500 \times (1,004)^n$.
Comme $C_n = V_n - 37500$, il vient :
$$C_n = 38500 \times (1,004)^n - 37500$$

\begin{enumerate}
  \setcounter{enumi}{3}
  \item \textbf{Résolution de l'objectif d'épargne (1,5 pt)} :
\end{enumerate}
On cherche le plus petit entier $n$ tel que $C_n \ge 25000$ :
$$38500 \times (1,004)^n - 37500 \ge 25000 \iff 38500 \times (1,004)^n \ge 62500 \iff (1,004)^n \ge \frac{62500}{38500} \approx 1,6234$$
En appliquant le logarithme népérien :
$$n \ln(1,004) \ge \ln\left(\frac{625}{385}\right) \iff n \ge \frac{\ln(1,6234)}{\ln(1,004)} \approx \frac{0,4845}{0,00399} \approx 121,3$$
Il faudra donc attendre \textbf{122 mois} (soit un peu plus de 10 ans) pour atteindre l'objectif de 25 000 €.

\begin{enumerate}
  \setcounter{enumi}{4}
  \item \textbf{Approche algorithmique (2 pts)} :
  \begin{enumerate}[label=\alph*.]
    \item Complétion de la fonction Python :
  \end{enumerate}
\end{enumerate}
\begin{verbatim}
def capital_mois(cible, versement, taux):
    n = 0
    c = 1000.0
    while c < cible:
        c = c * (1 + taux) + versement
        n = n + 1
    return n
\end{verbatim}
\begin{enumerate}
  \setcounter{enumi}{4}
  \begin{enumerate}[label=\alph*.]
    \setcounter{enumii}{1}
    \item L'appel \texttt{capital\_mois(25000, 150, 0.004)} renvoie exactement l'entier \textbf{122}, ce qui valide par calcul itératif le résultat théorique obtenu à la question précédente.
  \end{enumerate}
\end{enumerate}
