Voici le Devoir Maison complet consacré à l'étude de la fonction logarithme népérien en classe de Terminale Spécialité Mathématiques. Ce sujet aborde l'ensemble des méthodes fondamentales : calcul de dérivée, étude des variations, limites aux bornes, croissances comparées et application du théorème des valeurs intermédiaires.


\vspace{0.35cm}\noindent\textbf{\large Devoir Maison — La fonction logarithme népérien}\par\vspace{0.15cm}

\exo{1}{Étude d'une fonction et application}

On considère la fonction $f$ définie sur l'intervalle $]0 ; +\infty[$ par :
$$f(x) = x - 2\ln(x)$$
On note $\mathcal{C}_f$ la courbe représentative de $f$ dans un repère orthogonal $(O, \vec{i}, \vec{j})$.

\textbf{Partie A : Étude d'une fonction auxiliaire}
\begin{enumerate}
  \item Soit $g$ la fonction définie sur $]0 ; +\infty[$ par $g(x) = x - 2$. Étudier le signe de $g(x)$ sur $]0 ; +\infty[$.
  \item Calculer la dérivée $f'$ de la fonction $f$ et vérifier que pour tout $x > 0$ :
\end{enumerate}
$$f'(x) = \frac{g(x)}{x}$$
\begin{enumerate}
  \item En déduire le sens de variation de la fonction $f$ sur $]0 ; +\infty[$ et préciser la valeur exacte de son extremum.
\end{enumerate}

\textbf{Partie B : Limites et branches infinies}
\begin{enumerate}
  \item Déterminer la limite de $f$ en $0^+$. Interpréter graphiquement ce résultat pour la courbe $\mathcal{C}_f$.
  \item Déterminer la limite de $f$ en $+\infty$ en factorisant l'expression par $x$.
  \item Rappeler le résultat de cours concernant la croissance comparée de $\ln(x)$ et $x$ en $+\infty$, puis vérifier que la courbe $\mathcal{C}_f$ admet une direction asymptotique simple.
\end{enumerate}


\vspace{0.35cm}\noindent\textbf{\large Corrigé détaillé}\par\vspace{0.15cm}

\partie{Partie A : Étude de la fonction auxiliaire et de $f$}

\begin{enumerate}
  \item \textbf{Signe de $g(x) = x - 2$ :}
  \item $g(x) \ge 0 \iff x \ge 2$.
  \item La fonction $g$ est strictement négative sur $]0 ; 2[$, nulle en $2$, et positive sur $]2 ; +\infty[$.
\end{enumerate}

\begin{enumerate}
  \item \textbf{Dérivée de $f$ :}
\end{enumerate}
La fonction $f$ est dérivable sur $]0 ; +\infty[$ en tant que somme de fonctions dérivables :
$$f'(x) = 1 - 2 \times \frac{1}{x} = 1 - \frac{2}{x} = \frac{x - 2}{x}$$
Pour tout $x > 0$, le dénominateur $x$ est strictement positif. Par conséquent, \textbf{le signe de $f'(x)$ est celui du numérateur $x - 2$}, c'est-à-dire de $g(x)$.

\begin{enumerate}
  \item \textbf{Tableau de variations et extremum :}
  \item Sur $]0 ; 2]$, $f'(x) \le 0$ : la fonction $f$ est \textbf{strictement décroissante}.
  \item Sur $[2 ; +\infty[$, $f'(x) \ge 0$ : la fonction $f$ est \textbf{strictement croissante}.
  \item Valeur de l'extremum (minimum) en $x = 2$ :
\end{enumerate}
$$f(2) = 2 - 2\ln(2) = 2 - \ln(4) \approx 2 - 1{,}386 = 0{,}614$$


\partie{Partie B : Limites et branches infinies}

\begin{enumerate}
  \item \textbf{Limite en $0^+$ :}
  \item On a $\lim_{x \to 0^+} x = 0$ et $\lim_{x \to 0^+} \ln(x) = -\infty$.
  \item Par produit, $\lim_{x \to 0^+} (-2\ln(x)) = +\infty$.
  \item Par somme, \textbf{$\lim_{x \to 0^+} f(x) = +\infty$}.
  \item \textbf{Interprétation graphique :} La droite d'équation $x = 0$ (l'axe des ordonnées) est asymptote verticale à la courbe $\mathcal{C}_f$.
\end{enumerate}

\begin{enumerate}
  \item \textbf{Limite en $+\infty$ :}
\end{enumerate}
Pour tout $x > 0$, on factorise par $x$ :
$$f(x) = x\left(1 - 2\frac{\ln(x)}{x}\right)$$
\begin{enumerate}
  \item Par croissances comparées, on sait que $\lim_{x \to +\infty} \frac{\ln(x)}{x} = 0$.
  \item Par conséquent, $\lim_{x \to +\infty} \left(1 - 2\frac{\ln(x)}{x}\right) = 1$.
  \item Comme $\lim_{x \to +\infty} x = +\infty$, on obtient par produit :
\end{enumerate}
\textbf{$\lim_{x \to +\infty} f(x) = +\infty$}.
