% ---------------------- EXERCICE 1 ----------------------
\exo{1}{Étude d'une fonction dérivée et applications}

\partie{Partie A -- Étude d'une fonction auxiliaire}

Soit $g$ la fonction définie sur l'intervalle $]0 ; +\infty[$ par :
\[
g(x) = x^3 - 2 + 2\ln(x)
\]

\begin{enumerate}
  \item Déterminer la limite de $g(x)$ en $0$ et interpréter graphiquement ce résultat.
  \item Déterminer la limite de $g(x)$ en $+\infty$.
  \item Calculer la dérivée $g'(x)$ pour tout $x \in ]0 ; +\infty[$ et étudier son signe.
  \item Dresser le tableau de variations complet de la fonction $g$.
  \item Démontrer que l'équation $g(x) = 0$ admet une unique solution $\alpha$ sur $]0 ; +\infty[$. Donner un encadrement de $\alpha$ d'amplitude $10^{-2}$.
  \item En déduire le signe de $g(x)$ suivant les valeurs de $x$ sur $]0 ; +\infty[$.
\end{enumerate}

\partie{Partie B -- Étude de la fonction principale}

Soit $f$ la fonction définie sur $]0 ; +\infty[$ par :
\[
f(x) = x - 2 + \frac{2\ln(x)}{x^2}
\]
On note $\mathcal{C}$ sa courbe représentative dans un repère orthonormé $(O ; \vec{i}, \vec{j})$.

\begin{enumerate}
  \item Déterminer la limite de $f(x)$ en $0$. Interpréter graphiquement ce résultat.
  \item Déterminer la limite de $f(x)$ en $+\infty$.
  \item Démontrer que la droite $\Delta$ d'équation $y = x - 2$ est asymptote oblique à la courbe $\mathcal{C}$ en $+\infty$, et préciser la position relative de $\mathcal{C}$ et de $\Delta$.
  \item Démontrer que pour tout $x \in ]0 ; +\infty[$, on a :
  \[
  f'(x) = \frac{g(x)}{x^3}
  \]
  \item En déduire le tableau de variations de la fonction $f$.
  \item Démontrer que l'équation $f(x) = 0$ admet exactement deux solutions sur $]0 ; +\infty[$, dont l'une est notée $\beta$. Vérifier que $1 < \beta < 1,5$.
\end{enumerate}
